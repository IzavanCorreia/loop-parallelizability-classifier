\newcommand{\program}[3]{
  \node[txt, font=\fs{7}\bfseries] at (#1, 3.45) {#2};
  \fill[ngram!18] (#1 - 0.06, 2.21) rectangle (#1 + 2.16, 2.5);
  \node[code] at (#1, 3.0) {t = 0};
  \node[code] at (#1, 2.65) {for x in xs:};
  \node[code] at (#1, 2.3) {\ \ #3};
  \node[code] at (#1, 1.95) {\ \ ys.append(t)};
}
